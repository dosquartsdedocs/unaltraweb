\newcommand{\PracticeFigure}[2]{%
  \par\begingroup
  \setlength{\parskip}{0pt}%
  \setbox\PracticeCaptionBox=\vbox{\hsize=\linewidth\captionsetup{type=figure}#1\par}%
  \setlength{\PracticeImageHeight}{\dimexpr\textheight-\ht\PracticeCaptionBox-\dp\PracticeCaptionBox-\PracticeCaptionGap-2pt\relax}%
  \ifdim\PracticeImageHeight<\baselineskip
    \PackageError{unaltraweb-practice}{Figure caption exceeds the usable page height}{Shorten the caption or split the figure.}%
  \fi
  \setkeys{Gin}{width=\linewidth,height=\PracticeImageHeight,keepaspectratio}%
  \sbox{\PracticeImageBox}{#2}%
  \setbox\PracticeFigureBox=\vbox{%
    \hsize=\linewidth\offinterlineskip
    \hbox to\linewidth{\hfil\usebox{\PracticeCaptionBox}\hfil}%
    \vskip\PracticeCaptionGap
    \hbox to\linewidth{\hfil\usebox{\PracticeImageBox}\hfil}%
  }%
  \setlength{\PracticeFigureHeight}{\dimexpr\ht\PracticeFigureBox+\dp\PracticeFigureBox+2pt\relax}%
  \Needspace{\PracticeFigureHeight}%
  \nointerlineskip\usebox{\PracticeFigureBox}\par
  \endgroup\addvspace{1em}%
}
